  {{- if .Values.apiserver.tlsOn }}
  kubectl get secret --namespace {{ .Release.Namespace }} {{ include ".helm.apiserverTLSSecret" . }} -o jsonpath="{.data.ca\.crt}" | base64 -d > ca.crt
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ .Release.Name }}-routr-apiserver 51907 &
  rctl agents get --cacert ca.crt
  {{- else }}
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ .Release.Name }}-routr-apiserver 51907 &
  rctl agents get --insecure
  {{- end }}
